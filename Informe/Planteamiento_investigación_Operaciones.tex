	% 2. Introducción
	\begin{frame}{Introducción}
		\begin{itemize}
			\item \textbf{Árbol Rojinegro (RBT):} Árbol binario de búsqueda auto-balanceado.
			\item \textbf{Problema de la Mutabilidad:} En sistemas concurrentes, el cambio de estado "in-place" genera race conditions.
			\item \textbf{Solución:} Persistencia funcional mediante inmutabilidad y reparto estructural.
		\end{itemize}
	\end{frame}
	
	% 3. Propiedades RBT
	\begin{frame}{Propiedades Fundamentales del RBT}
		Para que un árbol sea Rojinegro, debe cumplir:
		\vfill
		\begin{enumerate}
			\item Todo nodo es o \textbf{Rojo} o \textbf{Negro}.
			\item La raíz es siempre \textbf{Negra}.
			\item Las hojas (nodos NIL) son \textbf{Negras}.
			\item Si un nodo es Rojo, sus hijos deben ser Negros.
			\item Cada ruta desde un nodo a sus hojas descendientes contiene el mismo número de nodos Negros.
		\end{enumerate}
	\end{frame}
	
	% 4. ¿Qué es la Persistencia?
	\begin{frame}{El Concepto de Persistencia}
		Una estructura de datos es \textbf{persistente} si siempre preserva su versión anterior cuando es modificada.
		\vfill
		\begin{block}{Tipos de Persistencia}
			\begin{itemize}
				\item \textbf{Parcial:} Solo se puede modificar la versión más reciente.
				\item \textbf{Total:} Todas las versiones pueden ser consultadas y modificadas.
			\end{itemize}
		\end{block}
	\end{frame}
	
	% 5. Structural Sharing
	\begin{frame}{Reparto Estructural (Structural Sharing)}
		¿Cómo logramos persistencia sin copiar todo el árbol?
		\vfill
		\begin{itemize}
			\item \textbf{Path Copying:} Solo se crean nuevos nodos para la ruta que va desde la raíz hasta el nodo modificado.
			\item Los nodos que no están en la ruta de modificación son compartidos por la versión antigua y la nueva.
			\item \textbf{Costo Espacial:} $O(\log n)$ por cada inserción o eliminación.
		\end{itemize}
	\end{frame}
	
	% 6. Haskell: Un Enfoque Puramente Funcional
	\begin{frame}{Haskell y Estructuras Puramente Funcionales}
		En Haskell, todos los datos son inmutables por definición.
		\vfill
		\begin{itemize}
			\item No existen los punteros modificables como en C.
			\item La recursión y el \textbf{Pattern Matching} sustituyen a los ciclos y condicionales complejos de balanceo.
		\end{itemize}
	\end{frame}
	
	% 7. Implementación de Okasaki
	\begin{frame}[fragile]{Haskell: Definición del Tipo de Dato}
		\begin{lstlisting}[language=Haskell]
			data Color = R | B
			data Tree a = E | T Color (Tree a) a (Tree a)
			
			-- Insercion persistente
			insert :: Ord a => a -> Tree a -> Tree a
			insert x s = makeBlack (ins s)
			where
			ins E = T R E x E
			ins (T col l y r)
			| x < y     = balance col (ins l) y r
			| x > y     = balance col l y (ins r)
			| otherwise = T col l y r
			makeBlack (T _ l y r) = T B l y r
		\end{lstlisting}
	\end{frame}
	
	% 8. La Función de Balanceo (Pattern Matching)
	\begin{frame}[fragile]{Haskell: El Elegante Balanceo de Okasaki}
		En lugar de rotaciones manuales, usamos pattern matching para identificar los 4 casos de desequilibrio:
		\begin{lstlisting}[language=Haskell]
			balance B (T R (T R a x b) y c) z d = T R (T B a x b) y (T B c z d)
			balance B (T R a x (T R b y c)) z d = T R (T B a x b) y (T B c z d)
			balance B a x (T R (T R b y c) z d) = T R (T B a x b) y (T B c z d)
			balance B a x (T R b y (T R c z d)) = T R (T B a x b) y (T B c z d)
			balance col l x r = T col l x r
		\end{lstlisting}
	\end{frame}
	
	% 9. Clojure: Inmutabilidad en la JVM
	\begin{frame}{Clojure: Persistencia Pragmática}
		Clojure utiliza \textbf{Persistent Data Structures} como base del lenguaje.
		\vfill
		\begin{itemize}
			\item Implementa \texttt{PersistentTreeMap} basado en Árboles Rojinegros.
			\item Utiliza una técnica llamada \textbf{Hashing} combinada con árboles para mayor eficiencia.
			\item Está diseñado para el manejo de estado en sistemas concurrentes masivos.
		\end{itemize}
	\end{frame}
	
	% 10. Clojure: Ejemplo de Uso
	\begin{frame}[fragile]{Clojure: Sorted Maps}
		\begin{lstlisting}[language=Lisp]
			;; Creacion de un mapa ordenado persistente
			(def t1 (sorted-map :a 1 :b 2))
			
			;; Generamos una nueva version (t2) sin alterar t1
			(def t2 (assoc t1 :c 3))
			
			(println t1) ; Output: {:a 1, :b 2}
			(println t2) ; Output: {:a 1, :b 2, :c 3}
		\end{lstlisting}
		\vfill
		\textit{Nota: Ambas versiones comparten los nodos de :a y :b en memoria.}
	\end{frame}
	
	% 11. Ventajas de la Persistencia
	\begin{frame}{Ventajas de la Persistencia}
		\begin{block}{Seguridad en Concurrencia}
			Múltiples hilos pueden leer el árbol sin necesidad de bloqueos (Locks), ya que los datos nunca cambian.
		\end{block}
		\vfill
		\begin{block}{Persistencia Histórica}
			Permite implementar funciones de "Undo/Redo" de forma eficiente simplemente manteniendo punteros a raíces antiguas.
		\end{block}
	\end{frame}
	
	% 12. Análisis de Complejidad
	\begin{frame}{Análisis de Complejidad}
		\begin{table}[]
			\centering
			\begin{tabular}{|l|c|c|}
				\hline
				\textbf{Operación} & \textbf{Tiempo} & \textbf{Espacio Adicional} \\ \hline
				Búsqueda           & $O(\log n)$     & $O(1)$                     \\ \hline
				Inserción          & $O(\log n)$     & $O(\log n)$                \\ \hline
				Eliminación        & $O(\log n)$     & $O(\log n)$                \\ \hline
				Copia (Versión)    & $O(1)$          & $O(1)$                     \\ \hline
			\end{tabular}
		\end{table}
		\vfill
		El costo espacial $O(\log n)$ es aceptable frente al beneficio de la inmutabilidad.
	\end{frame}
